                const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'tournament_store', 'main');

                if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
                    await signInWithCustomToken(auth, __initial_auth_token);
                } else {
                    await signInAnonymously(auth);
                }

                onSnapshot(docRef, (docSnap) => {
                    if (docSnap.exists()) {
                        const data = docSnap.data() || {};
                        if (Array.isArray(data.categories)) window.appState.categories = data.categories;
                        if (data.pairs && typeof data.pairs === 'object') window.appState.pairs = data.pairs;
                        if (Array.isArray(data.news)) window.appState.news = data.news;
                        if (data.matches && typeof data.matches === 'object') window.appState.matches = data.matches;

                        if (typeof renderHomeCategoriesGrids === 'function') renderHomeCategoriesGrids();
                        if (typeof renderCategoryDropdowns === 'function') renderCategoryDropdowns();
                        if (typeof renderInicioNews === 'function') renderInicioNews();
                        if (typeof renderNews === 'function') renderNews();
                        if (typeof renderFixtureForCategory === 'function') renderFixtureForCategory(window.appState.selectedCategory);
                    }
                }, err => console.log("Firebase sync omitted or error:", err));

                window._cloudPersist = async function() {
                    const safeConfig = { ...(window.appState.config || {}) };
                    delete safeConfig.adminPass;
                    await setDoc(docRef, {
                        categories: window.appState.categories || [],
                        pairs: window.appState.pairs || {},
                        news: window.appState.news || [],
                        matches: window.appState.matches || {},
                        config: safeConfig,
                        updatedAt: new Date().toISOString()
                    }, { merge: false });
                };
            }
        } catch (e) {
            console.log("Modo local activo (sin servicios externos).");
        }

                    id: Date.now(),
                    title,
                    date: "Reciente",
                    content,
                    image: imgSrc || "https://images.unsplash.com/photo-1554068865-24cecd4e34b8?w=600&auto=format&fit=crop&q=80"
                };

                window.appState.news.unshift(newObj);

                document.getElementById('input-news-image').value = '';

                const reader = new FileReader();
                reader.onload = function(e) {
                    createNewsObj(e.target.result);
                };
                reader.readAsDataURL(fileInput.files[0]);
            } else {
                createNewsObj(urlImage);
            }

        function isWinner(scoreA, scoreB) {
            if (!scoreA || !scoreB) return false;
            if (scoreA === "W.O.") return true;
            if (scoreB === "W.O.") return false;

            const setsA = scoreA.trim().split(' ').map(Number);
            const setsB = scoreB.trim().split(' ').map(Number);
            let wA = 0, wB = 0;
            for (let i = 0; i < Math.max(setsA.length, setsB.length); i++) {
                if ((setsA[i] || 0) > (setsB[i] || 0)) wA++;
                else if ((setsB[i] || 0) > (setsA[i] || 0)) wB++;
            }
            return wA > wB;
        }

        async function generateFEPFixtureLogic(cat, showAlert = false, forceReset = false) {
            const pairs = Array.isArray(window.appState.pairs[cat]) ? window.appState.pairs[cat] : [];
            window.appState.pairs[cat] = pairs;

            const oldMatches = Array.isArray(window.appState.matches[cat]) ? window.appState.matches[cat] : [];
            const oldById = {};
            if (!forceReset) {
                oldMatches.forEach(m => {
                    if (m && m.id) oldById[m.id] = { ...m };
                });
            }

            const newMatches = [];
            const zones = "ABCDEFGH".split('');

            const addMatch = (id, zone, round, t1, t2, court = "PP1", extra = {}) => {
                const old = oldById[id];
                newMatches.push({
                    id, cat, zone, round,
                    t1: old && old.t1 ? old.t1 : t1,
                    t2: old && old.t2 ? old.t2 : t2,
                    score1: old?.score1 || "",
                    score2: old?.score2 || "",
                    status: old?.status || "Programado",
                    datetime: old?.datetime || "A definir",
                    court: old?.court || court
                });
                Object.keys(extra || {}).forEach(key => {
                    if (extra[key] !== undefined) newMatches[newMatches.length - 1][key] = extra[key];
                });
            };

            const pairName = (p) => p ? `${p.j1} / ${p.j2}` : "Libre";

            if (pairs.length < 2) {
                window.appState.matches[cat] = [];
                renderFixtureForCategory(cat);
                await window.persistData();
                if (showAlert) alert("Cargá al menos 2 parejas para poder armar la llave.");
                return;
            }

            const active = pairs.map(p => ({ ...p }));
            if (active.length % 2 !== 0) {
                active.push({ id: "bye_" + Date.now(), j1: "BYE", j2: "Libre", isBye: true });
            }

            let zoneCount = 0;
            for (let i = 0; i < active.length; i += 2) {
                const code = zones[zoneCount] || `Z${zoneCount + 1}`;
                const p1 = active[i];
                const p2 = active[i + 1];
                const bye = !!p2.isBye;

                addMatch(
                    `z_${code}`,
                    `Zona ${code} (1er Partido)`,
                    "Clasificatorios",
                    pairName(p1),
                    bye ? "BYE / Libre" : pairName(p2),
                    bye ? "-" : "PP1",
                    {
                        code,
                        score1: bye ? "W.O." : undefined,
                        score2: bye ? "0 0" : undefined,
                        status: bye ? "Finalizado" : undefined,
                        datetime: bye ? "Automático" : undefined
                    }
                );
                zoneCount++;
            }

            if (pairs.length === 14) {
                addMatch("oct_1", "Octavos A", "Octavos", "C2", "F2", "PP1");
                addMatch("oct_2", "Octavos B", "Octavos", "D1", "G2", "PP2");
                addMatch("oct_3", "Octavos C", "Octavos", "E1", "B1", "PP3");
                addMatch("oct_4", "Octavos D", "Octavos", "A1", "F1", "PP4");
                addMatch("oct_5", "Octavos E", "Octavos", "D2", "C1", "SPORT P 1");
                addMatch("oct_6", "Octavos F", "Octavos", "E2", "G1", "SPORT P 2");

                addMatch("qf_1", "Cuartos A", "Cuartos", "A2", "Ganador Octavos A", "PP1");
                addMatch("qf_2", "Cuartos B", "Cuartos", "Ganador Octavos B", "Ganador Octavos C", "PP2");
                addMatch("qf_3", "Cuartos C", "Cuartos", "Ganador Octavos D", "Ganador Octavos E", "PP3");
                addMatch("qf_4", "Cuartos D", "Cuartos", "Ganador Octavos F", "B2", "PP4");

                addMatch("sf_1", "Semis A", "Semis", "Ganador Cuartos A", "Ganador Cuartos B", "PP1");
                addMatch("sf_2", "Semis B", "Semis", "Ganador Cuartos C", "Ganador Cuartos D", "PP2");
            }
            else if (pairs.length === 9 || pairs.length === 10) {
                addMatch("oct_1", "Octavos A", "Octavos", "C2", "D2", "PP1");
                addMatch("oct_2", "Octavos B", "B2", "E2", "PP2");

                addMatch("qf_1", "Cuartos A", "Cuartos", "A1", "Ganador Octavos A", "PP1");
                addMatch("qf_2", "Cuartos B", "Cuartos", "B1", "A2", "PP2");
                addMatch("qf_3", "Cuartos C", "Cuartos", "C1", "Ganador Octavos B", "PP3");
                addMatch("qf_4", "Cuartos D", "Cuartos", "D1", "E1", "PP4");

                addMatch("sf_1", "Semis A", "Semis", "Ganador Cuartos A", "Ganador Cuartos B", "PP1");
                addMatch("sf_2", "Semis B", "Semis", "Ganador Cuartos C", "Ganador Cuartos D", "PP2");
            }
            else if (pairs.length === 16) {
                addMatch("qf_1", "Cuartos A", "Cuartos", "Ganador Zona A (A1)", "Ganador Zona B (B1)", "PP1");
                addMatch("qf_2", "Cuartos B", "Cuartos", "Ganador Zona C (C1)", "Ganador Zona D (D1)", "PP2");
                addMatch("qf_3", "Cuartos C", "Cuartos", "Ganador Zona E (E1)", "Ganador Zona F (F1)", "PP3");
                addMatch("qf_4", "Cuartos D", "Cuartos", "Ganador Zona G (G1)", "Ganador Zona H (H1)", "PP4");
                addMatch("sf_1", "Semis A", "Semis", "Ganador Cuartos A", "Ganador Cuartos B", "PP1");
                addMatch("sf_2", "Semis B", "Semis", "Ganador Cuartos C", "Ganador Cuartos D", "PP2");
            }
            else {
                const qfCount = Math.min(4, Math.ceil(zoneCount / 2));
                for (let q = 0; q < qfCount; q++) {
                    const a = zones[q * 2];
                    const b = zones[q * 2 + 1];
                    if (!a) break;
                    addMatch(
                        `qf_${q + 1}`,
                        `Cuartos ${String.fromCharCode(65 + q)}`,
                        "Cuartos",
                        `Ganador Zona ${a} (${a}1)`,
                        b && q * 2 + 1 < zoneCount ? `Ganador Zona ${b} (${b}1)` : "Libre",
                        `PP${(q % 4) + 1}`
                    );
                }
                addMatch("sf_1", "Semis A", "Semis", "Ganador Cuartos A", "Ganador Cuartos B", "PP1");
                addMatch("sf_2", "Semis B", "Semis", "Ganador Cuartos C", "Ganador Cuartos D", "PP2");
            }

            addMatch("fn_1", "Final", "Final", "Ganador Semis A", "Ganador Semis B", "PP1");

            const replaceToken = (token, value) => {
                newMatches.forEach(target => {
                    if (typeof target.t1 === "string" && target.t1 === token) target.t1 = value;
                    if (typeof target.t2 === "string" && target.t2 === token) target.t2 = value;
                });
            };

            newMatches.forEach(m => {
                if (m.status !== "Finalizado" || !m.score1 || !m.score2) return;

                const winner = isWinner(m.score1, m.score2) ? m.t1 : m.t2;
                const loser = isWinner(m.score1, m.score2) ? m.t2 : m.t1;

                if (m.code) {
                    replaceToken(`${m.code}1`, winner);
                    replaceToken(`${m.code}2`, loser);
                    replaceToken(`Ganador Zona ${m.code} (${m.code}1)`, winner);
                }

                const round = m.round;
                if (round === "Octavos" || round === "Cuartos" || round === "Semis") {
                    newMatches.forEach(target => {
                        if (target.t1 === `Ganador ${m.zone}`) target.t1 = winner;
                        if (target.t2 === `Ganador ${m.zone}`) target.t2 = winner;
                    });
                }
            });

            window.appState.matches[cat] = newMatches;
            renderFixtureForCategory(cat);
            await window.persistData();

            if (showAlert) alert(`¡Fixture de ${cat} actualizado con éxito!`);
        }

            const cat = window.appState.selectedCategory;

            renderModalPairsList();
            document.getElementById('modal-manage-pairs').classList.remove('hidden');

        function renderModalPairsList() {
            const cat = window.appState.selectedCategory;
            const pairs = window.appState.pairs[cat] || [];
            document.getElementById('count-pairs-cat').innerText = pairs.length;

            const list = document.getElementById('modal-pairs-list');
            if (pairs.length === 0) {
                list.innerHTML = `<p class="text-xs text-slate-500 py-2 text-center">Aún no agregaste parejas a esta categoría.</p>`;
                return;
            }

            list.innerHTML = pairs.map((p, idx) => `
                <div class="bg-navy-800 border border-navy-700 p-2.5 rounded-lg flex items-center justify-between text-xs">
                    <div class="flex items-center gap-2">
                        <span class="font-bold text-slate-500">#${idx+1}</span>
                        <span class="font-bold text-white">${p.j1} / ${p.j2}</span>
                    </div>
                    <button onclick="deletePairFromCat('${p.id}')" class="text-rose-400 hover:text-rose-300 font-bold px-2 py-1 bg-rose-500/10 rounded border border-rose-500/20">
                        Eliminar
                    </button>
                </div>
            `).join('');
        }

        async function saveNewPair() {
            if (!window.appState.isAdmin) {
                alert("Necesitás ingresar como administrador.");
                return;
            }

            const cat = window.appState.selectedCategory;
            const j1El = document.getElementById('input-j1-name');
            const j2El = document.getElementById('input-j2-name');
            const photoEl = document.getElementById('input-pair-photo');
            const phoneEl = document.getElementById('input-pair-phone');

            const j1 = j1El ? j1El.value.trim() : '';
            const j2 = j2El ? j2El.value.trim() : '';
            const photo = photoEl ? photoEl.value.trim() : '';
            const phone = phoneEl ? phoneEl.value.trim() : '';

            if (!j1 || !j2) {
                alert("Ingresá el nombre de ambos jugadores.");
                return;
            }

            if (!window.appState.pairs[cat]) window.appState.pairs[cat] = [];

            const pairId = 'pair_' + Date.now() + '_' + Math.random().toString(36).slice(2, 8);

            window.appState.pairs[cat].push({
                id: pairId,
                j1, j2, photo, phone
            });

            renderModalPairsList();
            renderFixtureForCategory(cat);

            if (j1El) j1El.value = '';
            if (j2El) j2El.value = '';
            if (photoEl) photoEl.value = '';
            if (phoneEl) phoneEl.value = '';

            await generateFEPFixtureLogic(cat, false, false);
            renderModalPairsList();
        }

        async function deletePairFromCat(pairId) {
            if (!window.appState.isAdmin) {
                alert("Necesitás ingresar como administrador.");
                return;
            }

            const cat = window.appState.selectedCategory;
            const before = window.appState.pairs[cat] || [];
            window.appState.pairs[cat] = before.filter(p => p.id !== pairId);

            renderModalPairsList();
            renderFixtureForCategory(cat);
            await generateFEPFixtureLogic(cat, false, false);
            renderModalPairsList();
        }

            const cat = window.appState.selectedCategory;
            const match = window.appState.matches[cat].find(m => m.id === matchId);
            if (!match) return;

            document.getElementById('edit-match-id').value = match.id;

            document.getElementById('match-team1-label').innerText = match.t1;

            document.getElementById('p1-set1').value = '';
            document.getElementById('p1-set2').value = '';
            document.getElementById('p1-set3').value = '';
            document.getElementById('p2-set1').value = '';
            document.getElementById('p2-set2').value = '';
            document.getElementById('p2-set3').value = '';

            if (match.score1) {
                const s1 = match.score1.split(' ');
                if (s1[0]) document.getElementById('p1-set1').value = s1[0];
                if (s1[1]) document.getElementById('p1-set2').value = s1[1];
                if (s1[2]) document.getElementById('p1-set3').value = s1[2];
            }

            if (match.score2) {
                const s2 = match.score2.split(' ');
                if (s2[0]) document.getElementById('p2-set1').value = s2[0];
                if (s2[1]) document.getElementById('p2-set2').value = s2[1];
                if (s2[2]) document.getElementById('p2-set3').value = s2[2];
            }

            if (match.court) {
                document.getElementById('select-match-court').value = match.court;
            }

            document.getElementById('input-match-status').value = match.status || 'Programado';

            document.getElementById('modal-edit-match').classList.remove('hidden');
        }

        function saveMatchDetails() {
            const cat = window.appState.selectedCategory;
            const matchId = document.getElementById('edit-match-id').value;
            const matchesList = window.appState.matches[cat];
            const match = matchesList.find(m => m.id === matchId);

            if (match) {
                const p1Set1 = document.getElementById('p1-set1').value.trim();
                const p1Set2 = document.getElementById('p1-set2').value.trim();
                const p1Set3 = document.getElementById('p1-set3').value.trim();

                const p2Set1 = document.getElementById('p2-set1').value.trim();
                const p2Set2 = document.getElementById('p2-set2').value.trim();
                const p2Set3 = document.getElementById('p2-set3').value.trim();

                const s1Array = [p1Set1, p1Set2, p1Set3].filter(x => x !== '');
                const s2Array = [p2Set1, p2Set2, p2Set3].filter(x => x !== '');

                const s1 = s1Array.join(' ');
                const s2 = s2Array.join(' ');

                const status = document.getElementById('input-match-status').value;
                const day = document.getElementById('select-match-day').value;
                const time = document.getElementById('select-match-time').value;
                const court = document.getElementById('select-match-court').value;

                match.score1 = s1;
                match.score2 = s2;
                match.datetime = `${day} - ${time}`;
                match.court = court;
                match.status = status;

                if (status === 'Finalizado' && s1 && s2) {
                    const winT1 = isWinner(s1, s2);
                    const winner = winT1 ? match.t1 : match.t2;
                    const loser = winT1 ? match.t2 : match.t1;

                    if (match.code) {
                        const code = match.code;
                        matchesList.forEach(m => {
                            if (m.t1.includes(`(${code}1)`)) m.t1 = winner;
                            if (m.t2.includes(`(${code}1)`)) m.t1 = winner;
                            if (m.t1.includes(`(${code}2)`)) m.t1 = loser;
                            if (m.t2.includes(`(${code}2)`)) m.t2 = loser;
                        });
                    } else if (match.zone) {
                        const matchZoneName = match.zone;
                        matchesList.forEach(m => {
                            if (m.t1.includes(`Ganador ${matchZoneName}`)) m.t1 = winner;
                            if (m.t2.includes(`Ganador ${matchZoneName}`)) m.t2 = winner;
                        });
                    }
                }

                renderFixtureForCategory(cat);
                window.persistData();
                closeModal('modal-edit-match');
            }
        }

                const hourStr = h < 10 ? `0${h}` : `${h}`;
                timesHTML += `<option value="${hourStr}:00 hs">${hourStr}:00 hs</option>`;
                timesHTML += `<option value="${hourStr}:30 hs">${hourStr}:30 hs</option>`;
            }
            select.innerHTML = timesHTML;
        }
